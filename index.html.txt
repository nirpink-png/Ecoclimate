    // -------------------------------------------------------------
    // Web Speech API / Voice Control Parsing
    // -------------------------------------------------------------
    function initSpeechRecognition() {
      const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
      if (!SpeechRecognition) {
        console.warn("Speech Recognition API not supported in this browser environment.");
        return;
      }

      recognition = new SpeechRecognition();
      recognition.continuous = false;
      recognition.interimResults = true;

      recognition.onstart = () => {
        isListening = true;
        updateMicUI(true);
        document.getElementById('voiceFeedbackContainer').classList.remove('hidden');
        document.getElementById('voiceTranscript').innerText = i18n[currentLang].listeningMsg;
      };

      recognition.onresult = (event) => {
        const transcript = Array.from(event.results)
          .map(result => result[0].transcript)
          .join('');
        
        document.getElementById('voiceTranscript').innerText = `"${transcript}"`;

        if (event.results[0].isFinal) {
          parseVoiceCommand(transcript.toLowerCase());
        }
      };

      recognition.onerror = (event) => {
        console.error("Speech Recognition Error:", event.error);
        stopVoiceControl();
      };

      recognition.onend = () => {
        stopVoiceControl();
      };
    }

    function toggleVoiceControl() {
      if (!recognition) {
        alert("Voice recognition is not supported on this browser. Try Chrome on Android or Safari on iOS.");
        return;
      }

      if (isListening) {
        recognition.stop();
      } else {
        // Set speech language dynamically based on current app language setting
        const langMap = { en: 'en-US', es: 'es-ES', fr: 'fr-FR', hi: 'hi-IN' };
        recognition.lang = langMap[currentLang] || 'en-US';
        recognition.start();
      }
    }

    function stopVoiceControl() {
      isListening = false;
      updateMicUI(false);
      setTimeout(() => {
        document.getElementById('voiceFeedbackContainer').classList.add('hidden');
      }, 3000);
    }

        micBtn.className = "pointer-events-auto bg-red-500 hover:bg-red-600 text-white rounded-full p-4 shadow-xl shadow-red-500/40 animate-bounce transition-all duration-300";
        micIcon.className = "fa-solid fa-microphone-slash text-2xl";
      } else {
        micBtn.className = "pointer-events-auto bg-green-500 hover:bg-green-600 text-white rounded-full p-4 shadow-xl shadow-green-500/30 flex items-center justify-center transition-all duration-300 transform active:scale-90";
        micIcon.className = "fa-solid fa-microphone text-2xl";
      }
    }

    function parseVoiceCommand(cmd) {
      // 1. Language Commands
      if (cmd.includes('spanish') || cmd.includes('español')) {
        changeLanguage('es');
        document.getElementById('langSelect').value = 'es';
      } else if (cmd.includes('french') || cmd.includes('français')) {
        changeLanguage('fr');
        document.getElementById('langSelect').value = 'fr';
      } else if (cmd.includes('hindi') || cmd.includes('हिंदी')) {
        changeLanguage('hi');
        document.getElementById('langSelect').value = 'hi';
      } else if (cmd.includes('english')) {
        changeLanguage('en');
        document.getElementById('langSelect').value = 'en';
      }

      // 2. Navigation / Highlight Commands
      if (cmd.includes('air quality') || cmd.includes('aqi') || cmd.includes('calidad del aire') || cmd.includes('qualité de l\'air') || cmd.includes('हवा')) {
        highlightAndScroll('aqiSection');
      } else if (cmd.includes('forecast') || cmd.includes('10 day') || cmd.includes('pronóstico') || cmd.includes('prévision') || cmd.includes('पूर्वाभ्यास')) {
        highlightAndScroll('forecastSection');
      }

      // 3. Theme Command
      if (cmd.includes('dark mode') || cmd.includes('light mode') || cmd.includes('modo oscuro') || cmd.includes('mode sombre')) {
        toggleTheme();
      }
    }

    function highlightAndScroll(elementId) {
      const el = document.getElementById(elementId);
      if (el) {
        el.scrollIntoView({ behavior: 'smooth', block: 'center' });
        el.classList.add('highlight-card');
        setTimeout(() => {
          el.classList.remove('highlight-card');
        }, 3000);
      }
    }
